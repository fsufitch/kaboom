AGENT: Kaboom monorepo test strategy (first step)

Goal
- Add fast, source-based unit tests (no build prerequisite) using Vitest in a pnpm workspace monorepo.
- Standardize each workspace package layout to: src/ + tests/.
- Update bundler entrypoints to only consume src/ so tests never get bundled into dist/.
- Add per-package Vitest configs and per-package devDependencies (using pnpm catalog:), so each package can run its own tests.
- Add placeholder “always passing” tests for @kaboom/proto and @kaboom/cli.
- Add an initial real test suite for @kaboom/engine covering classic bishop captures (bishop_capture.ts).

Non-goals (for this step)
- Do not build/bundle as a prerequisite to running tests.
- Do not write exhaustive test coverage across the engine; only the first slice (bishop_capture) plus placeholders.
- Do not refactor code behavior; only move files and adjust imports/paths/configs to preserve behavior.
- Do not introduce a shared “root vitest projects” config; keep configs per package.

Acceptance criteria
1) Repo installs cleanly with pnpm.
2) Repo structure:
   - packages/engine/src/** and packages/engine/tests/**
   - packages/proto/src/** and packages/proto/tests/**
   - packages/cli/src/** and packages/cli/tests/**
3) Bundling still works:
   - pnpm -C packages/proto run build
   - pnpm -C packages/engine run build
   - pnpm -C packages/cli run build
4) Testing works without building:
   - pnpm -C packages/engine run test   (this must generate proto code as a prereq)
   - pnpm -C packages/proto run test    (placeholder test passes)
   - pnpm -C packages/cli run test      (placeholder test passes)
   - pnpm -r run test                   (works without needing --if-present)
5) Engine bishop_capture tests:
   - Valid capture enumeration across diagonals (blocked after first piece).
   - resolveToEffects produces expected stateChanges (pieceMoved + pieceCaptured).
   - Illegal capture throws IllegalMoveError.

Create: packages/engine/vitest.config.ts
Content (ESM-safe):

- Allow Vite/Vitest to load files outside the package root (proto package) via server.fs.allow.
- Inline workspace deps pattern for @kaboom/* to ensure TS sources are transformed.

Use this exact file content:

  import path from 'node:path';
  import { fileURLToPath } from 'node:url';
  import { defineConfig } from 'vitest/config';

  const engineRoot = fileURLToPath(new URL('.', import.meta.url));
  const repoRoot = fileURLToPath(new URL('../..', import.meta.url));
  const protoIndex = fileURLToPath(new URL('../proto/src/index.ts', import.meta.url));

  export default defineConfig({
    root: engineRoot,
    resolve: {
      alias: [
        { find: /^@kaboom\/engine$/, replacement: path.join(engineRoot, 'src/index.ts') },
        { find: /^@kaboom\/engine\/(.*)$/, replacement: path.join(engineRoot, 'src/$1') },
        { find: /^@kaboom\/proto$/, replacement: protoIndex },
      ],
    },
    server: {
      fs: {
        allow: [repoRoot],
      },
    },
    test: {
      environment: 'node',
      include: ['tests/**/*.test.ts'],
      exclude: ['dist/**', 'node_modules/**'],
      deps: { inline: [/^@kaboom\//] },
      coverage: {
        provider: 'v8',
        reportsDirectory: path.join(repoRoot, 'coverage', 'engine'),
      },
    },
  });

Step 11: Add initial real tests for bishop_capture in @kaboom/engine
Create: packages/engine/tests/classic/bishop_capture.test.ts

This suite should:
- Build a minimal GameSnapshot with exactly one board and a handful of pieces.
- Assert BishopCaptureResolver.validMoves returns exactly the capturable destinations (first enemy piece on each diagonal; blocked afterwards; friendly blocks stop).
- Assert resolveToEffects returns a single Effect with two StateChanges: pieceMoved then pieceCaptured.
- Assert resolveToEffects throws IllegalMoveError for an illegal capture (e.g., “capture to empty square”).

Use this exact content (adjust only if compilation errors reveal minor naming differences from generated proto TS):

  import { describe, expect, it } from 'vitest';

  import {
    ChessBoard,
    ChessColor,
    ChessPiece,
    ChessPieceKind,
    GameSnapshot,
    Move,
    Variant,
  } from '@kaboom/proto';

  import { IllegalMoveError, movesEqual } from '@kaboom/engine/base';
  import { BishopCaptureResolver } from '@kaboom/engine/classic/bishop_capture';

  const BOARD_ID = 'board-1';

  function mkBoard(): ChessBoard {
    return ChessBoard.create({
      id: BOARD_ID,
      rows: 8,
      columns: 8,
      activeColor: ChessColor.WHITE,
    });
  }

  function mkPiece(
    id: string,
    kind: ChessPieceKind,
    color: ChessColor,
    row: number,
    column: number,
  ): ChessPiece {
    return ChessPiece.create({
      id,
      kind,
      color,
      place: {
        boardId: BOARD_ID,
        boardPosition: { row, column },
      },
    });
  }

  function mkSnapshot(pieces: ChessPiece[]): GameSnapshot {
    return GameSnapshot.create({
      properties: { id: 'game-1', variant: Variant.CLASSIC },
      boards: [mkBoard()],
      pieces,
    });
  }

  function mkBishopCaptureMove(fromRow: number, fromCol: number, toRow: number, toCol: number): Move {
    return Move.create({
      classicMove: {
        bishop: {
          capture: {
            from: { boardId: BOARD_ID, boardPosition: { row: fromRow, column: fromCol } },
            to: { boardId: BOARD_ID, boardPosition: { row: toRow, column: toCol } },
          },
        },
      },
    });
  }

  describe('BishopCaptureResolver (classic)', () => {
    it('enumerates only the first capturable opponent on each diagonal, and stops at blockers', () => {
      const bishop = mkPiece('wb', ChessPieceKind.BISHOP, ChessColor.WHITE, 3, 3);

      // NE: first enemy at (5,5) is capturable; piece behind at (6,6) must not be.
      const neTarget = mkPiece('bp-ne', ChessPieceKind.PAWN, ChessColor.BLACK, 5, 5);
      const neBehind = mkPiece('bq-ne-behind', ChessPieceKind.QUEEN, ChessColor.BLACK, 6, 6);

      // NW: enemy at (6,0) is capturable (diagonal up-left)
      const nwTarget = mkPiece('br-nw', ChessPieceKind.ROOK, ChessColor.BLACK, 6, 0);

      // SW: immediate enemy at (2,2) is capturable
      const swTarget = mkPiece('bn-sw', ChessPieceKind.KNIGHT, ChessColor.BLACK, 2, 2);

      // SE: friendly at (1,5) blocks; enemy behind at (0,6) must not be capturable
      const seBlock = mkPiece('wp-block', ChessPieceKind.PAWN, ChessColor.WHITE, 1, 5);
      const seBehind = mkPiece('bp-se-behind', ChessPieceKind.PAWN, ChessColor.BLACK, 0, 6);

      const gs = mkSnapshot([bishop, neTarget, neBehind, nwTarget, swTarget, seBlock, seBehind]);

      const moves = BishopCaptureResolver.validMoves(gs, bishop.id);

      const expected = [
        mkBishopCaptureMove(3, 3, 5, 5),
        mkBishopCaptureMove(3, 3, 6, 0),
        mkBishopCaptureMove(3, 3, 2, 2),
      ];

      expect(moves).toHaveLength(3);
      for (const e of expected) {
        expect(moves.some((m) => movesEqual(m, e))).toBe(true);
      }

      expect(moves.some((m) => movesEqual(m, mkBishopCaptureMove(3, 3, 6, 6)))).toBe(false);
      expect(moves.some((m) => movesEqual(m, mkBishopCaptureMove(3, 3, 0, 6)))).toBe(false);
    });

    it('resolves a legal capture into pieceMoved + pieceCaptured state changes', () => {
      const bishop = mkPiece('wb', ChessPieceKind.BISHOP, ChessColor.WHITE, 3, 3);
      const target = mkPiece('bp', ChessPieceKind.PAWN, ChessColor.BLACK, 5, 5);

      const gs = mkSnapshot([bishop, target]);
      const move = mkBishopCaptureMove(3, 3, 5, 5);

      const effects = BishopCaptureResolver.resolveToEffects(gs, move);

      expect(effects).toHaveLength(1);
      const effect = effects[0];
      expect(effect).toBeDefined();

      expect(effect!.stateChanges).toHaveLength(2);

      const sc0 = effect!.stateChanges[0];
      const sc1 = effect!.stateChanges[1];

      expect(sc0?.pieceMoved?.pieceId).toBe(bishop.id);
      expect(sc0?.pieceMoved?.to).toEqual({ row: 5, column: 5 });

      expect(sc1?.pieceCaptured?.pieceId).toBe(target.id);
    });

    it('throws IllegalMoveError for an illegal capture move (e.g., capture to empty square)', () => {
      const bishop = mkPiece('wb', ChessPieceKind.BISHOP, ChessColor.WHITE, 3, 3);
      const gs = mkSnapshot([bishop]);

      const illegal = mkBishopCaptureMove(3, 3, 4, 4);

      expect(() => BishopCaptureResolver.resolveToEffects(gs, illegal)).toThrow(IllegalMoveError);
    });
  });
